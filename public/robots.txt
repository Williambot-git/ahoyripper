# General crawlers (User-agent: *) — applies to all bots not listed above.
# IMPORTANT: Disallow rules MUST come BEFORE Allow rules in the same
# User-agent block. Googlebot uses first-matching-rule precedence (not
# "most specific wins"), so "Allow: /" before "Disallow: /src/" means
# Googlebot never sees the /src/ block. Place more-specific rules first.
# Specific crawlers (GPTBot, ClaudeBot, etc.) have their own blocks above
# and do NOT fall through to this section.
User-agent: *
